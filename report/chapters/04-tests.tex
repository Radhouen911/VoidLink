\section{Testing Strategy}

\subsection{Testing Approach}

VoidLink employs a multi-layered testing strategy to verify both functional correctness and security properties:

\textbf{Unit Testing:}
\begin{itemize}
    \item Cryptographic functions (encryption, decryption, signing)
    \item Key management operations
    \item Database helper functions
    \item Utility functions and data transformations
\end{itemize}

\textbf{Integration Testing:}
\begin{itemize}
    \item API endpoint functionality
    \item Database operations and transactions
    \item WebSocket message handling
    \item Authentication flows
\end{itemize}

\textbf{End-to-End Testing:}
\begin{itemize}
    \item Complete user workflows (registration to messaging)
    \item Real-time communication scenarios
    \item Offline message delivery
    \item Multi-user interactions
\end{itemize}

\textbf{Security Testing:}
\begin{itemize}
    \item Authentication bypass attempts
    \item Encryption/decryption verification
    \item Session management security
    \item Input validation and injection prevention
\end{itemize}

\subsection{Testing Infrastructure}

\textbf{Backend Testing:}
\begin{itemize}
    \item Node.js test scripts with real cryptographic operations
    \item Direct database access for verification
    \item WebSocket client simulation
    \item Automated test suite execution
\end{itemize}

\textbf{Frontend Testing:}
\begin{itemize}
    \item Playwright for end-to-end browser automation
    \item Headless browser testing
    \item Visual regression testing
    \item Cross-browser compatibility testing
\end{itemize}

\textbf{Test Environment:}
\begin{itemize}

\section{System Limitations}

\subsection{Security Limitations}

\textbf{No Forward Secrecy:}
\begin{itemize}
    \item \textit{Issue:} Static key pairs used for all messages
    \item \textit{Impact:} Compromise of private key exposes all past messages
    \item \textit{Mitigation:} None in current implementation
    \item \textit{Future Work:} Implement Double Ratchet algorithm
\end{itemize}

\textbf{Weak Passphrase Vulnerability:}
\begin{itemize}
    \item \textit{Issue:} Simple SHA-512 hash for key derivation
    \item \textit{Impact:} Weak passphrases vulnerable to brute force
    \item \textit{Mitigation:} User education on strong passphrases
    \item \textit{Future Work:} Implement PBKDF2 or Argon2 key stretching
\end{itemize}

\textbf{No Key Verification Enforcement:}
\begin{itemize}
    \item \textit{Issue:} Manual key fingerprint verification optional
    \item \textit{Impact:} Vulnerable to man-in-the-middle during key exchange
    \item \textit{Mitigation:} Key fingerprints available for manual verification
    \item \textit{Future Work:} Implement Safety Numbers like Signal
\end{itemize}

\textbf{Metadata Exposure:}
\begin{itemize}
    \item \textit{Issue:} Server logs communication patterns
    \item \textit{Impact:} Who communicates with whom is visible
    \item \textit{Mitigation:} None in current architecture
    \item \textit{Future Work:} Implement metadata-resistant routing
\end{itemize}

\subsection{Functional Limitations}

\textbf{No Group Messaging:}
\begin{itemize}
    \item \textit{Reason:} Deliberate design choice for simplicity
    \item \textit{Impact:} Only one-to-one conversations supported
    \item \textit{Workaround:} None available
    \item \textit{Future Work:} Implement group key management
\end{itemize}

\textbf{No Multi-Device Synchronization:}
\begin{itemize}
    \item \textit{Reason:} Complexity of secure synchronization
    \item \textit{Impact:} Each device requires separate login
    \item \textit{Workaround:} Cloud backup enables key recovery
    \item \textit{Future Work:} Implement device linking protocol
\end{itemize}

\textbf{No Message Editing:}
\begin{itemize}
    \item \textit{Reason:} Immutability simplifies security model
    \item \textit{Impact:} Sent messages cannot be modified
    \item \textit{Workaround:} Send correction message
    \item \textit{Future Work:} Implement edit history with signatures
\end{itemize}

\textbf{No Message Deletion:}
\begin{itemize}
    \item \textit{Reason:} Server-side deletion conflicts with zero-trust
    \item \textit{Impact:} Messages persist in database
    \item \textit{Workaround:} Soft delete hides from user interface
    \item \textit{Future Work:} Implement client-side deletion with sync
\end{itemize}

\textbf{No File Sharing:}
\begin{itemize}
    \item \textit{Reason:} Not implemented in current version
    \item \textit{Impact:} Only text messages supported
    \item \textit{Workaround:} None available
    \item \textit{Future Work:} Implement encrypted file transfer
\end{itemize}

\subsection{Performance Limitations}

\textbf{Client-Side Encryption Overhead:}
\begin{itemize}
    \item \textit{Issue:} All crypto operations on client device
    \item \textit{Impact:} Battery drain on mobile devices
    \item \textit{Mitigation:} Efficient NaCl implementation
    \item \textit{Measurement:} 1-2ms per message (acceptable)
\end{itemize}

\textbf{No Server-Side Search:}
\begin{itemize}
    \item \textit{Issue:} Encrypted messages cannot be searched on server
    \item \textit{Impact:} Search limited to loaded messages
    \item \textit{Mitigation:} Client-side search in message history
    \item \textit{Future Work:} Implement encrypted search index
\end{itemize}

\textbf{Message History Loading:}
\begin{itemize}
    \item \textit{Issue:} Large conversations require multiple queries
    \item \textit{Impact:} Pagination needed for performance
    \item \textit{Mitigation:} Infinite scroll with 30-message batches
    \item \textit{Measurement:} 20-30ms per batch (acceptable)
\end{itemize}

\subsection{Usability Limitations}

\textbf{Passphrase Management:}
\begin{itemize}
    \item \textit{Issue:} Users must remember separate passphrase
    \item \textit{Impact:} Lost passphrase means lost messages
    \item \textit{Mitigation:} Clear warnings during registration
    \item \textit{Future Work:} Implement social recovery mechanisms
\end{itemize}

\textbf{Key Verification Complexity:}
\begin{itemize}
    \item \textit{Issue:} Manual fingerprint comparison required
    \item \textit{Impact:} Most users skip verification
    \item \textit{Mitigation:} Simplified fingerprint display
    \item \textit{Future Work:} QR code scanning for verification
\end{itemize}

\textbf{No Account Recovery:}
\begin{itemize}
    \item \textit{Issue:} Lost credentials mean lost account
    \item \textit{Impact:} No password reset mechanism
    \item \textit{Mitigation:} Cloud backup enables key recovery
    \item \textit{Trade-off:} Security vs. convenience
\end{itemize}

\textbf{Forward Secrecy Implementation:}
\begin{itemize}
    \item Implement Double Ratchet algorithm (Signal Protocol)
    \item Generate ephemeral keys for each message
    \item Automatic key rotation after N messages or time period

\end{itemize}

\textbf{Key Stretching:}
\begin{itemize}
    \item Replace SHA-512 with PBKDF2 or Argon2
